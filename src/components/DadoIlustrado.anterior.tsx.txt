import { memo, useEffect, useRef, useState } from 'react';
import type { ReactNode } from 'react';
import { View, useWindowDimensions } from 'react-native';
import Svg, { Circle, Ellipse, G, Path, Rect } from 'react-native-svg';
import { ajustesActuales } from '../ajustesBase';
import { PAPEL, TINTA } from '../diseno';
import { sonar } from '../sonido';

// El dado del juego, dibujado como una ilustración: un cubo en tres cuartos con contorno de tinta.
// Se ven tres caras: la de arriba es el resultado y las dos laterales, otras caras del dado.
// (La versión anterior en 3D está guardada en Dado3D.anterior.tsx.txt)

const DURACION_TIRADA = 1500; // ms
const PAUSA_FINAL = 500; // ms quieto mostrando el resultado
const REBOTES = 3;
const CAMBIO_CARAS = 90; // ms entre cambios de cara mientras rueda

export interface TiradaEnCurso {
  id: number;
  valor: number;
  inclinacion?: number;
}

// Aspecto de una cara: color de fondo y, si lo hay, un dibujo (caja de 100 x 100); sin dibujo lleva los puntos
export interface CaraVisual {
  fondo: string;
  dibujo?: ReactNode;
}

export const CARAS_CLASICAS: CaraVisual[] = Array.from({ length: 6 }, () => ({ fondo: PAPEL }));

// Caras laterales que se ven junto a la de arriba (nunca la opuesta: las opuestas suman 7)
const LATERALES: Record<number, [number, number]> = {
  1: [2, 3],
  2: [6, 3],
  3: [2, 6],
  4: [5, 6],
  5: [1, 4],
  6: [5, 3],
};

const PUNTOS: Record<number, [number, number][]> = {
  1: [[50, 50]],
  2: [[28, 28], [72, 72]],
  3: [[26, 26], [50, 50], [74, 74]],
  4: [[28, 28], [72, 28], [28, 72], [72, 72]],
  5: [[26, 26], [74, 26], [50, 50], [26, 74], [74, 74]],
  6: [[28, 24], [72, 24], [28, 50], [72, 50], [28, 76], [72, 76]],
};

function sombrear(hex: string, f: number): string {
  const n = parseInt(hex.slice(1), 16);
  const c = (d: number) => Math.round(((n >> d) & 255) * f);
  return `rgb(${c(16)}, ${c(8)}, ${c(0)})`;
}

// Una cara plana (100 x 100) con su dibujo o sus puntos
function Cara({ valor, visual, luz }: { valor: number; visual: CaraVisual; luz: number }) {
  return (
    <>
      <Rect x={0} y={0} width={100} height={100} rx={10} fill={sombrear(visual.fondo, luz)} />
      {visual.dibujo ?? PUNTOS[valor].map(([x, y], i) => <Circle key={i} cx={x} cy={y} r={9} fill={TINTA} />)}
      <Rect x={0} y={0} width={100} height={100} rx={10} fill="none" stroke={TINTA} strokeWidth={3.4} />
    </>
  );
}

// Proyección de cada cara del cubo (caja de 300 x 300)
const ARRIBA = 'matrix(1 0.55 -1 0.55 150 40)';
const IZQUIERDA = 'matrix(1 0.55 0 1.12 50 95)';
const DERECHA = 'matrix(1 -0.55 0 1.12 150 150)';

const Cubo = memo(function Cubo({
  valor,
  caras,
  size,
}: {
  valor: number;
  caras: CaraVisual[];
  size: number;
}) {
  const [izq, der] = LATERALES[valor] ?? [2, 3];
  const vis = (v: number) => caras[v - 1] ?? caras[0];
  return (
    <Svg width={size} height={size} viewBox="0 0 300 300" style={{ overflow: 'visible' }}>
      <G transform={ARRIBA}>
        <Cara valor={valor} visual={vis(valor)} luz={1} />
      </G>
      <G transform={IZQUIERDA}>
        <Cara valor={izq} visual={vis(izq)} luz={0.92} />
      </G>
      <G transform={DERECHA}>
        <Cara valor={der} visual={vis(der)} luz={0.82} />
      </G>
    </Svg>
  );
});

function SombraSuelo({ size, rebote = 0, x = 0 }: { size: number; rebote?: number; x?: number }) {
  return (
    <Svg
      width={size}
      height={size * 0.14}
      viewBox="0 0 300 42"
      style={{ position: 'absolute', left: x, top: size * 0.86 }}
    >
      <Ellipse cx={150} cy={21} rx={74 * (1 - 0.35 * rebote)} ry={8} fill={TINTA} opacity={0.14 * (1 - 0.5 * rebote)} />
    </Svg>
  );
}

// ---------- Dado quieto ----------

type PropsReposo = {
  valor: number;
  size: number;
  inclinacion?: number;
  caras?: CaraVisual[];
};

export function DadoReposo({ valor, size, inclinacion = 0, caras = CARAS_CLASICAS }: PropsReposo) {
  return (
    <View style={{ width: size, height: size }}>
      <SombraSuelo size={size} />
      <View style={{ transform: [{ rotate: `${inclinacion * 12}deg` }] }}>
        <Cubo valor={valor} caras={caras} size={size} />
      </View>
    </View>
  );
}

// ---------- Dado lanzándose ----------

type PropsAnimado = {
  valor: number;
  size: number;
  onTerminar: () => void;
  inclinacion?: number; // giro con el que se queda
  desde?: { valor: number; inclinacion: number }; // cómo estaba antes de lanzarlo
  entrada?: 'lado' | 'centro'; // entra volando desde un lado o se lanza en su sitio
  caras?: CaraVisual[];
};

export function DadoAnimado({
  valor,
  size,
  onTerminar,
  inclinacion = 0,
  desde,
  entrada = 'lado',
  caras = CARAS_CLASICAS,
}: PropsAnimado) {
  const { width: ancho } = useWindowDimensions();
  const alTerminar = useRef(onTerminar);
  alTerminar.current = onTerminar;
  const rapido = ajustesActuales().reducirAnimaciones;
  const duracion = rapido ? 500 : DURACION_TIRADA;

  const [p, setP] = useState(0); // progreso, de 0 a 1
  const [mostrada, setMostrada] = useState(desde?.valor ?? 1 + Math.floor(Math.random() * 6));
  const [azar] = useState(() => ({ lado: Math.random() < 0.5 ? -1 : 1, giro: (Math.random() < 0.5 ? -1 : 1) * 2 }));

  useEffect(() => {
    // Suena cuando el dado toca la mesa por primera vez
    const golpe = setTimeout(() => sonar('dado'), duracion * 0.16);
    const inicio = Date.now();
    let ultimoCambio = 0;
    let cuadro = 0;
    let espera: ReturnType<typeof setTimeout> | undefined;
    const paso = () => {
      const t = Date.now() - inicio;
      const progreso = Math.min(t / duracion, 1);
      setP(progreso);
      // Mientras rueda, van pasando caras; al final se queda en la buena
      if (progreso < 0.78 && t - ultimoCambio > CAMBIO_CARAS) {
        ultimoCambio = t;
        setMostrada((v) => ((v + 1 + Math.floor(Math.random() * 4)) % 6) + 1);
      } else if (progreso >= 0.78) {
        setMostrada(valor);
      }
      if (progreso < 1) cuadro = requestAnimationFrame(paso);
      else espera = setTimeout(() => alTerminar.current(), PAUSA_FINAL);
    };
    cuadro = requestAnimationFrame(paso);
    return () => {
      clearTimeout(golpe);
      cancelAnimationFrame(cuadro);
      if (espera) clearTimeout(espera);
    };
  }, [valor]);

  // Botes cada vez más bajos, un giro que se frena y, si entra de lado, un deslizamiento
  const resto = Math.pow(1 - p, 1.6);
  const rebote = Math.abs(Math.sin(Math.PI * REBOTES * p)) * resto;
  const y = -rebote * size * 0.35;
  const x = entrada === 'lado' ? azar.lado * ancho * 0.6 * Math.pow(1 - p, 2) : 0;
  const giro = azar.giro * 180 * Math.pow(1 - p, 2.4) + inclinacion * 12 * p;
  const escala = 1 + 0.05 * rebote;

  return (
    <View style={{ width: size, height: size, opacity: entrada === 'lado' ? Math.min(1, p / 0.08) : 1 }}>
      <SombraSuelo size={size} rebote={rebote} x={x} />
      <View style={{ transform: [{ translateX: x }, { translateY: y }, { rotate: `${giro}deg` }, { scale: escala }] }}>
        <Cubo valor={mostrada} caras={caras} size={size} />
      </View>
    </View>
  );
}
